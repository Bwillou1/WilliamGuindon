Exemple d'Intégration Python (Pour Agents IA & RAG)

import urllib.request
import json

# Option A : Chargement direct du corpus complet via l'API REST
url = "https://williamguindon.me/api/v1/corpus.json"
req = urllib.request.Request(url, headers={"User-Agent": "AIAgent/1.0"})
with urllib.request.urlopen(req) as response:
    corpus = json.loads(response.read().decode("utf-8"))

print(f"Pages indexées : {corpus['statistics']['total_pages']}")
print(f"Nombre total de mots : {corpus['statistics']['total_words']}")

# Recherche plein texte instantanée dans toutes les pages
query = "cadmium"
matching_pages = [
    page for page in corpus["site_pages"]
    if query.lower() in page["full_text"].lower()
]
print(f"Pages mentionnant '{query}' : {len(matching_pages)}")
